```python
from gplearn.genetic import SymbolicTransformer

symbolic_transformer = SymbolicTransformer(
    population_size=1000,  # 种群大小
    generations=20,        # 进化代数
    tournament_size=20,    # 锦标赛选择的大小
    function_set=('add', 'sub', 'mul', 'div', 'sqrt', 'log', 'abs'),  # 允许的函数
    parsimony_coefficient=0.01,  # 简约系数，控制表达式的复杂度
    max_samples=0.9,        # 每次拟合时使用的样本比例
    verbose=1,             # 打印进度
    random_state=42
)
```

1）generations：遗传算法的迭代代数，一般取30-50代，任务越复杂则越大，同时消耗算力资源也越大。 

2）population_size：初始种群大小，也就是第一代进入“生存”竞赛的公式数量，通常为纯随机生成，经过设定好的代数更迭后选出“胜者”。  

3）function_set：遗传编程中使用的函数集合，如加法、减法、乘法、除法等。需要注意 的是，简单的加减乘除是肯定不够的，实战中我们一般会增加如自相关系数、滚动相关性、 峰谷值、峰谷位置等因素 

4）n_components：最后返回的特征数量，一般设为10或20即可。 

5）metric：用于评估新特征与目标变量相关性的指标，如 pearson （预测任务）或 spearman（分类任务）。 6）parsimony_coefficient：用于控制生成表达式复杂度的参数，一般为0.01。该数值越小意味着公式越“复杂”。 

7）max_samples：用于训练的样本比例，小样本数据可用1.0即全部数据，大样本数据建议0.70-0.90，可加快运算速度。 

8）hall_of_fame：中文意为“名人堂”，代表着每一代中进入最终决赛的数量，也就是最优秀个体。优秀的定义就是新特征与Y的相关性较高。

### 构建遗传算法模型
```python
# 感谢粉丝提供的函数集
def gen_function_set():

    windows = [5, 20, 60, 120]
    function_set = []
    for rolling_window in windows:
        def ts_delay(X):
            shifted = np.roll(X, rolling_window)
            return shifted


        function_name = f'ts_delay_{rolling_window}'
        globals()[function_name] = make_function(function=ts_delay, name=f'ts_delay_{rolling_window}', arity=1)
        function_set.append(function_name)


        def ts_corr(arr1, arr2):
            # print(rolling_window)
            if len(arr1) < rolling_window or len(arr2) < rolling_window:
                return np.zeros_like(arr1)
            else:
                window1 = sliding_window_view(arr1, window_shape=rolling_window)
                window2 = sliding_window_view(arr2, window_shape=rolling_window)

                mean1 = np.mean(window1, axis=1)
                mean2 = np.mean(window2, axis=1)
                std1 = np.std(window1, axis=1)
                std2 = np.std(window2, axis=1)

                cov = np.mean(window1 * window2, axis=1) - mean1 * mean2
                with np.errstate(divide='ignore', invalid='ignore'):
                    corr = cov / (std1 * std2)
                    corr[np.isnan(corr) | np.isinf(corr)] = 0
                return np.concatenate([np.zeros(rolling_window - 1), corr])


        function_name = f'ts_corr_{rolling_window}'
        globals()[function_name] = make_function(function=ts_corr, name=f'ts_corr_{rolling_window}', arity=2)
        function_set.append(function_name)


        def ts_cov(arr1, arr2):
            if len(arr1) < rolling_window or len(arr2) < rolling_window:
                return np.zeros_like(arr1)
            else:
                window1 = sliding_window_view(arr1, window_shape=rolling_window)
                window2 = sliding_window_view(arr2, window_shape=rolling_window)

                mean1 = np.mean(window1, axis=1)
                mean2 = np.mean(window2, axis=1)
                std1 = np.std(window1, axis=1)
                std2 = np.std(window2, axis=1)

                cov = np.mean(window1 * window2, axis=1) - mean1 * mean2
                cov[np.isnan(cov) | np.isinf(cov)] = 0
                return np.concatenate([np.zeros(rolling_window - 1), cov])


        function_name = f'ts_cov_{rolling_window}'
        globals()[function_name] = make_function(function=ts_cov, name=f'ts_cov_{rolling_window}', arity=2)
        function_set.append(function_name)


        def ts_max(arr):
            if len(arr) < rolling_window:
                return np.zeros_like(arr)
            else:
                rolling_max = np.max(sliding_window_view(arr, window_shape=rolling_window), axis=1)
                return np.concatenate([np.zeros(rolling_window - 1), rolling_max])


        function_name = f'ts_max_{rolling_window}'
        globals()[function_name] = make_function(function=ts_max, name=f'ts_max_{rolling_window}', arity=1)
        function_set.append(function_name)


        def ts_min(arr):
            if len(arr) < rolling_window:
                return np.zeros_like(arr)
            else:
                rolling_min = np.min(sliding_window_view(arr, window_shape=rolling_window), axis=1)
                return np.concatenate([np.zeros(rolling_window - 1), rolling_min])


        function_name = f'ts_min_{rolling_window}'
        globals()[function_name] = make_function(function=ts_min, name=f'ts_min_{rolling_window}', arity=1)
        function_set.append(function_name)


        def ts_maxarg(arr):
```

### Inception模块
```python
class InceptionModule(nn.Module):
    def __init__(self, in_channels, out_channels):
        super(InceptionModule, self).__init__()
        # 4个分支的卷积层
        self.branch1 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch2 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=5, padding=2),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch3 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=7, padding=3),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch4 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=9, padding=4),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        # 残差连接
        self.residual = nn.Sequential(
            nn.Conv1d(in_channels, out_channels * 4, kernel_size=1),
            nn.BatchNorm1d(out_channels * 4)
        )
        # 池化层
        self.pool = nn.MaxPool1d(kernel_size=2, stride=2)  # 最大池化层
        self.relu = nn.ReLU()

    def forward(self, x):
        # 4个分支的输出
        out1 = self.branch1(x)
        out2 = self.branch2(x)
        out3 = self.branch3(x)
        out4 = self.branch4(x)
        # 拼接4个分支的输出
        out = torch.cat([out1, out2, out3, out4], dim=1)
        # 残差连接
        residual = self.residual(x)
        out += residual
        out = self.relu(out)
        # 池化层
        out = self.pool(out)  # 池化后的输出
        return out
```

### Attention模块
```python
class SelfAttention(nn.Module):
    def __init__(self, embed_size):
        super(SelfAttention, self).__init__()
        self.embed_size = embed_size
        # 定义Q、K、V的线性变换
        self.query = nn.Linear(embed_size, embed_size)
        self.key = nn.Linear(embed_size, embed_size)
        self.value = nn.Linear(embed_size, embed_size)
        self.softmax = nn.Softmax(dim=2)

    def forward(self, x):
        # x的形状: (batch_size, seq_length, embed_size)
        Q = self.query(x)  # (batch_size, seq_length, embed_size)
        K = self.key(x)  # (batch_size, seq_length, embed_size)
        V = self.value(x)  # (batch_size, seq_length, embed_size)

        # 计算注意力分数
        attention_scores = torch.bmm(Q, K.transpose(1, 2))  # (batch_size, seq_length, seq_length)
        attention_scores = self.softmax(attention_scores)

        # 加权求和
        out = torch.bmm(attention_scores, V)  # (batch_size, seq_length, embed_size)
        return out
```

### 接入BiGRU
```python
class BiGRUInceptionAttentionBlock(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers, dropout=0.5):
        super(BiGRUInceptionAttentionBlock, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers

        # 定义双向GRU层
        self.gru = nn.GRU(input_size, hidden_size, num_layers, batch_first=True, bidirectional=True)

        # 定义Inception模块
        self.inception = InceptionModule(hidden_size * 2, hidden_size // 2)  # 输入通道数为hidden_size * 2（双向GRU的输出）

        # 定义Self-Attention层
        self.attention = SelfAttention((hidden_size // 2) * 4)  # Inception模块的输出通道数为 (hidden_size // 2) * 4

        # 定义Dropout层
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # 初始化隐藏状态
        h0 = torch.zeros(self.num_layers * 2, x.size(0), self.hidden_size).to(x.device)

        # 前向传播通过GRU
        out, _ = self.gru(x, h0)  # out的形状: (batch_size, seq_length, hidden_size * 2)

        # 将GRU的输出转置为(batch_size, hidden_size * 2, seq_length)以适应1D卷积
        out = out.permute(0, 2, 1)

        # 前向传播通过Inception模块
        out = self.inception(out)  # out的形状: (batch_size, (hidden_size // 2) * 4, seq_length // 2)

        # 将Inception模块的输出转置为(batch_size, seq_length // 2, (hidden_size // 2) * 4)以适应Attention层
        out = out.permute(0, 2, 1)

        # 前向传播通过Self-Attention层
        out = self.attention(out)  # out的形状: (batch_size, seq_length // 2, (hidden_size // 2) * 4)

        # 应用Dropout
        out = self.dropout(out)

        return out
```

### 汇总基础单元
```python
class BiGRUInceptionAttentionModel(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers, output_size, dropout=0.5):
        super(BiGRUInceptionAttentionModel, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers

        # 定义三个BiGRU + Inception + Attention模块
        self.block1 = BiGRUInceptionAttentionBlock(input_size, hidden_size, num_layers, dropout)
        self.block2 = BiGRUInceptionAttentionBlock((hidden_size // 2) * 4, hidden_size, num_layers, dropout)
        self.block3 = BiGRUInceptionAttentionBlock((hidden_size // 2) * 4, hidden_size, num_layers, dropout)

        # 定义全连接层
        self.fc = nn.Linear((hidden_size // 2) * 4, output_size)  # 最终输出维度

    def forward(self, x):
        # 前向传播通过三个模块
        out = self.block1(x)
        out = self.block2(out)
        out = self.block3(out)

        # 全局平均池化
        out = torch.mean(out, dim=1)  # out的形状: (batch_size, (hidden_size // 2) * 4)

        # 全连接层
        out = self.fc(out)  # out的形状: (batch_size, output_size)

        return out
```

```python
def gen_function_set():

    windows = [5, 20, 60, 120]
    function_set = []
    for rolling_window in windows:
        def ts_delay(X):
            shifted = np.roll(X, rolling_window)
            return shifted


        function_name = f'ts_delay_{rolling_window}'
        globals()[function_name] = make_function(function=ts_delay, name=f'ts_delay_{rolling_window}', arity=1)
        function_set.append(function_name)


        def ts_corr(arr1, arr2):
            # print(rolling_window)
            if len(arr1) < rolling_window or len(arr2) < rolling_window:
                return np.zeros_like(arr1)
            else:
                window1 = sliding_window_view(arr1, window_shape=rolling_window)
                window2 = sliding_window_view(arr2, window_shape=rolling_window)

                mean1 = np.mean(window1, axis=1)
                mean2 = np.mean(window2, axis=1)
                std1 = np.std(window1, axis=1)
                std2 = np.std(window2, axis=1)

                cov = np.mean(window1 * window2, axis=1) - mean1 * mean2
                with np.errstate(divide='ignore', invalid='ignore'):
                    corr = cov / (std1 * std2)
                    corr[np.isnan(corr) | np.isinf(corr)] = 0
                return np.concatenate([np.zeros(rolling_window - 1), corr])


        function_name = f'ts_corr_{rolling_window}'
        globals()[function_name] = make_function(function=ts_corr, name=f'ts_corr_{rolling_window}', arity=2)
        function_set.append(function_name)


        def ts_cov(arr1, arr2):
            if len(arr1) < rolling_window or len(arr2) < rolling_window:
                return np.zeros_like(arr1)

                window1 = sliding_window_view(arr1, window_shape=rolling_window)
                window2 = sliding_window_view(arr2, window_shape=rolling_window)

                mean1 = np.mean(window1, axis=1)
                mean2 = np.mean(window2, axis=1)
                std1 = np.std(window1, axis=1)
                std2 = np.std(window2, axis=1)

                cov = np.mean(window1 * window2, axis=1) - mean1 * mean2
                cov[np.isnan(cov) | np.isinf(cov)] = 0
                return np.concatenate([np.zeros(rolling_window - 1), cov])


        function_name = f'ts_cov_{rolling_window}'

        def ts_kurt(arr):
            if len(arr) < rolling_window:
                return np.zeros_like(arr)
            else:
                windows = sliding_window_view(arr, window_shape=rolling_window)
                stds = np.std(windows, axis=1)
                means = np.mean(windows, axis=1, keepdims=True)
                deviations = windows - means
                mu4 = np.mean(deviations ** 4, axis=1)
                stds[stds == 0] = 1
                kurtosis_values = mu4 / stds ** 4 - 3
                return np.concatenate([np.zeros(rolling_window - 1), kurtosis_values])

                zscore_values = (windows - means) / stds

                return np.concatenate([np.zeros(rolling_window - 1), zscore_values])

                return np.concatenate([np.zeros(rolling_window - 1), weighted_means])
```

```python
# 2. 定义Inception模块
class InceptionModule(nn.Module):
    def __init__(self, in_channels, out_channels):
        super(InceptionModule, self).__init__()
        # 4个分支的卷积层
        self.branch1 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch2 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=5, padding=2),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch3 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=7, padding=3),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        self.branch4 = nn.Sequential(
            nn.Conv1d(in_channels, out_channels, kernel_size=9, padding=4),
            nn.BatchNorm1d(out_channels),
            nn.ReLU()
        )
        # 残差连接
        self.residual = nn.Sequential(
            nn.Conv1d(in_channels, out_channels * 4, kernel_size=1),
            nn.BatchNorm1d(out_channels * 4)
        )
        # 池化层
        self.pool = nn.MaxPool1d(kernel_size=2, stride=2)  # 最大池化层
        self.relu = nn.ReLU()

    def forward(self, x):
        # 4个分支的输出
        out1 = self.branch1(x)
        out2 = self.branch2(x)
        out3 = self.branch3(x)
        out4 = self.branch4(x)
        # 拼接4个分支的输出
        out = torch.cat([out1, out2, out3, out4], dim=1)
        # 残差连接
        residual = self.residual(x)
        out += residual
        out = self.relu(out)
        # 池化层
        out = self.pool(out)  # 池化后的输出
        return out


# 3. 定义Self-Attention层
class SelfAttention(nn.Module):
    def __init__(self, embed_size):
        super(SelfAttention, self).__init__()
        self.embed_size = embed_size
        # 定义Q、K、V的线性变换
        self.query = nn.Linear(embed_size, embed_size)
        self.key = nn.Linear(embed_size, embed_size)
        self.value = nn.Linear(embed_size, embed_size)
        self.softmax = nn.Softmax(dim=2)

    def forward(self, x):
        # x的形状: (batch_size, seq_length, embed_size)
        Q = self.query(x)  # (batch_size, seq_length, embed_size)
        K = self.key(x)  # (batch_size, seq_length, embed_size)
        V = self.value(x)  # (batch_size, seq_length, embed_size)

        # 计算注意力分数
        attention_scores = torch.bmm(Q, K.transpose(1, 2))  # (batch_size, seq_length, seq_length)
        attention_scores = self.softmax(attention_scores)

        # 加权求和
        out = torch.bmm(attention_scores, V)  # (batch_size, seq_length, embed_size)
        return out


# 4. 定义BiGRU + Inception + Attention模块
class BiGRUInceptionAttentionBlock(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers, dropout=0.5):
        super(BiGRUInceptionAttentionBlock, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers

        # 定义双向GRU层
        self.gru = nn.GRU(input_size, hidden_size, num_layers, batch_first=True, bidirectional=True)

        # 定义Inception模块
        self.inception = InceptionModule(hidden_size * 2, hidden_size // 2)  # 输入通道数为hidden_size * 2（双向GRU的输出）

        # 定义Self-Attention层
        self.attention = SelfAttention((hidden_size // 2) * 4)  # Inception模块的输出通道数为 (hidden_size // 2) * 4

        # 定义Dropout层
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # 初始化隐藏状态
        h0 = torch.zeros(self.num_layers * 2, x.size(0), self.hidden_size).to(x.device)

        # 前向传播通过GRU
        out, _ = self.gru(x, h0)  # out的形状: (batch_size, seq_length, hidden_size * 2)

        # 将GRU的输出转置为(batch_size, hidden_size * 2, seq_length)以适应1D卷积
        out = out.permute(0, 2, 1)

        # 前向传播通过Inception模块
        out = self.inception(out)  # out的形状: (batch_size, (hidden_size // 2) * 4, seq_length // 2)

        # 将Inception模块的输出转置为(batch_size, seq_length // 2, (hidden_size // 2) * 4)以适应Attention层
        out = out.permute(0, 2, 1)

        # 前向传播通过Self-Attention层
        out = self.attention(out)  # out的形状: (batch_size, seq_length // 2, (hidden_size // 2) * 4)

        # 应用Dropout
        out = self.dropout(out)

        return out


# 5. 定义完整的模型（循环三次BiGRU + Inception + Attention模块）
class BiGRUInceptionAttentionModel(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers, output_size, dropout=0.5):
        super(BiGRUInceptionAttentionModel, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers

        # 定义三个BiGRU + Inception + Attention模块
        self.block1 = BiGRUInceptionAttentionBlock(input_size, hidden_size, num_layers, dropout)
        self.block2 = BiGRUInceptionAttentionBlock((hidden_size // 2) * 4, hidden_size, num_layers, dropout)
        self.block3 = BiGRUInceptionAttentionBlock((hidden_size // 2) * 4, hidden_size, num_layers, dropout)

        # 定义全连接层
        self.fc = nn.Linear((hidden_size // 2) * 4, output_size)  # 最终输出维度

    def forward(self, x):
        # 前向传播通过三个模块
        out = self.block1(x)
        out = self.block2(out)
        out = self.block3(out)

        # 全局平均池化
        out = torch.mean(out, dim=1)  # out的形状: (batch_size, (hidden_size // 2) * 4)

        # 全连接层
        out = self.fc(out)  # out的形状: (batch_size, output_size)

        return out
```
